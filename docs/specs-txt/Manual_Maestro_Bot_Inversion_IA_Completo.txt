4. Módulo 2: Filtro Macro y Noticias Cisne Negro (macro_news.py)
Monitorea el índice VIX de Wall Street y procesa titulares de Reuters en tiempo real con el modelo FinBERT corriendo 100% en local para vetar operaciones durante pánico.
| import yfinance as yf
import feedparser
from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch

# Inicialización de FinBERT Local
tokenizer = AutoTokenizer.from_pretrained("ProsusAI/finbert")
model = AutoModelForSequenceClassification.from_pretrained("ProsusAI/finbert")

def check_vix_circuit_breaker():
    """Veta compras si VIX > 25 o si subió más de 10% en el día."""
    try:
        vix = yf.Ticker("^VIX").history(period="2d")
        if len(vix) < 2:
            return True, "Datos VIX insuficientes"
        vix_actual = vix['Close'].iloc[-1]
        vix_prev = vix['Close'].iloc[-2]
        cambio_pct = (vix_actual - vix_prev) / vix_prev
        
        if vix_actual > 25.0:
            return False, f"VETO: VIX en nivel de pánico ({vix_actual:.2f} > 25.0)"
        if cambio_pct > 0.10:
            return False, f"VETO: Subida abrupta de volatilidad VIX (+{cambio_pct*100:.1f}%)"
        return True, f"VIX OK ({vix_actual:.2f})"
    except Exception as e:
        return False, f"Error leyendo VIX: {e}"

def get_finbert_sentiment(ticker):
    """Descarga noticias RSS de Reuters/Yahoo y calcula el Sentiment Score (-1 a +1)."""
    PALABRAS_CRITICAS = ["war", "sanction", "bank run", "fraud", "default", "bankruptcy", "sec probe"]
    url = f"https://finance.yahoo.com/rss/headline?s={ticker}"
    feed = feedparser.parse(url)
    
    if not feed.entries:
        return True, 0.0, "Sin noticias recientes (Neutro)"
        
    scores = []
    for entry in feed.entries[:5]:  # Evalúa las últimas 5 noticias
        titulo = entry.title.lower()
        if any(w in titulo for w in PALABRAS_CRITICAS):
            return False, -1.0, f"VETO CISNE NEGRO: Palabra crítica detectada en '{entry.title}'"
            
        inputs = tokenizer(entry.title, return_tensors="pt", padding=True, truncation=True)
        outputs = model(**inputs)
        probs = torch.nn.functional.softmax(outputs.logits, dim=-1)[0]
        # Probs: [Positive, Negative, Neutral]
        score = probs[0].item() - probs[1].item()
        scores.append(score)
        
    avg_score = sum(scores) / len(scores) if scores else 0.0
    if avg_score < -0.35:
        return False, avg_score, f"VETO SENTIMIENTO: FinBERT Score negativo ({avg_score:.2f} < -0.35)"
    return True, avg_score, f"Sentimiento OK ({avg_score:.2f})"

5. Módulo 3: Rastreo de Ballenas e Insiders (insider_smartmoney.py)
Scraper de compras del Form 4 de la SEC (Cluster Buys de directores) y detector matemático de volumen intradiario anómalo (RVOL > 2.5).
| import yfinance as yf
import requests
import xml.etree.ElementTree as ET

def check_unusual_volume(ticker):
    """Detecta acumulación institucional oculta: RVOL > 2.0 y vela compresión."""
    data = yf.Ticker(ticker).history(period="30d")
    if len(data) < 20:
        return False, 0.0
    vol_actual = data['Volume'].iloc[-1]
    vol_medio = data['Volume'].iloc[-21:-1].mean()
    rvol = vol_actual / vol_medio if vol_medio > 0 else 0
    
    # RVOL > 2.0 indica entrada de manos institucionales
    es_acumulacion = rvol >= 2.0
    return es_acumulacion, rvol

def check_sec_form4_cluster(ticker):
    """Audita compras en mercado abierto (Código P) por ejecutivos en SEC EDGAR."""
    headers = {"User-Agent": "QuantitativeTradingBot research@investingbot.com"}
    url = f"https://www.sec.gov/cgi-bin/browse-edgar?action=getcompany&CIK={ticker}&type=4&output=atom"
    try:
        res = requests.get(url, headers=headers, timeout=10)
        if res.status_code == 200:
            root = ET.fromstring(res.content)
            # Detección de filings de Form 4 recientes en los últimos 7 días
            entradas_form4 = len(root.findall('{http://www.w3.org/2005/Atom}entry'))
            tiene_insiders = entradas_form4 >= 2
            return tiene_insiders, f"Insiders detectados: {entradas_form4} filings Form 4"
        return False, "Sin respuesta SEC"
    except Exception as e:
        return False, f"Error SEC API: {e}"

6. Módulo 4: Los 7 Filtros de Validación Técnica (technical_filters.py)
Filtros técnicos de tendencia, calendario de earnings, cálculo de volatilidad dinámica con ATR y confirmación de R/B 1:2.
| import yfinance as yf
import pandas as pd
import numpy as np

def calculate_atr(df, period=14):
    high_low = df['High'] - df['Low']
    high_close = np.abs(df['High'] - df['Close'].shift())
    low_close = np.abs(df['Low'] - df['Close'].shift())
    ranges = pd.concat([high_low, high_close, low_close], axis=1)
    true_range = np.max(ranges, axis=1)
    return true_range.rolling(period).mean()

def evaluate_7_filters(ticker):
    """Aplica el protocolo estricto de 7 capas antes de autorizar orden de compra."""
    stock = yf.Ticker(ticker)
    spy = yf.Ticker("SPY").history(period="60d")
    df = stock.history(period="60d")
    
    if len(df) < 50 or len(spy) < 50:
        return False, {}

    # 1. Filtro Calendario de Balances (Earnings)
    try:
        cal = stock.calendar
        if cal is not None and not cal.empty:
            proximo_balance = pd.to_datetime(cal.iloc[0, 0])
            dias_para_balance = (proximo_balance - pd.Timestamp.now()).days
            if 0 <= dias_para_balance <= 7:
                return False, {"motivo": f"VETO EARNINGS: Presenta en {dias_para_balance} días"}
    except:
        pass

    # 2. Régimen de Mercado SPY (Mayor a EMA 50)
    spy['EMA50'] = spy['Close'].ewm(span=50, adjust=False).mean()
    if spy['Close'].iloc[-1] < spy['EMA50'].iloc[-1]:
        return False, {"motivo": "VETO MERCADO: SPY por debajo de su EMA 50"}

    # 3. Tendencia del Activo y Medias
    df['EMA20'] = df['Close'].ewm(span=20, adjust=False).mean()
    df['EMA50'] = df['Close'].ewm(span=50, adjust=False).mean()
    precio_actual = df['Close'].iloc[-1]
    if not (precio_actual > df['EMA20'].iloc[-1] > df['EMA50'].iloc[-1]):
        return False, {"motivo": "VETO TÉCNICO: Estructura no alcista (Precio < EMA20)"}

    # 4. Fuerza Relativa (RS) vs SPY en 30 ruedas
    retorno_stock = (precio_actual / df['Close'].iloc[-30]) - 1
    retorno_spy = (spy['Close'].iloc[-1] / spy['Close'].iloc[-30]) - 1
    if retorno_stock <= retorno_spy:
        return False, {"motivo": "VETO RS: Activo no supera el rendimiento de SPY"}

    # 5. Volatilidad Dinámica (ATR) y Stop-Loss
    atr = calculate_atr(df).iloc[-1]
    stop_loss = precio_actual - (1.5 * atr)
    pct_riesgo = (precio_actual - stop_loss) / precio_actual
    
    # Límite máximo de riesgo: 3.0%
    if pct_riesgo > 0.035:
        stop_loss = precio_actual * 0.97  # Ajuste al 3% estricto
        pct_riesgo = 0.030

    take_profit = precio_actual + (precio_actual - stop_loss) * 2.0  # R/B 1:2

    return True, {
        "ticker": ticker,
        "precio_entrada": round(precio_actual, 2),
        "stop_loss": round(stop_loss, 2),
        "take_profit": round(take_profit, 2),
        "pct_riesgo": round(pct_riesgo * 100, 2)
    }

7. Módulo 5: Conector IOL y Ejecución Automática (broker_iol.py)
Gestiona autenticación mediante OAuth2, obtiene el Bearer Token oficial, consulta saldo líquido y envía órdenes con Límite en Contado Inmediato.
| import requests
import time
from config import IOL_USERNAME, IOL_PASSWORD, IOL_API_URL

class IOLClient:
    def __init__(self):
        self.token = None
        self.token_expiry = 0
        self.login()

    def login(self):
        """Obtiene el Bearer Token de autenticación de IOL."""
        url = f"{IOL_API_URL}/token"
        payload = {
            "username": IOL_USERNAME,
            "password": IOL_PASSWORD,
            "grant_type": "password"
        }
        res = requests.post(url, data=payload)
        if res.status_code == 200:
            data = res.json()
            self.token = data["access_token"]
            self.token_expiry = time.time() + data.get("expires_in", 900) - 60
            print("Autenticación exitosa en IOL.")
        else:
            raise Exception(f"Fallo login IOL: {res.text}")

    def get_headers(self):
        if time.time() >= self.token_expiry:
            self.login()
        return {"Authorization": f"Bearer {self.token}"}

    def get_saldo_disponible(self):
        """Consulta saldo disponible en pesos para operar."""
        url = f"{IOL_API_URL}/api/v2/estadocuenta"
        res = requests.get(url, headers=self.get_headers())
        if res.status_code == 200:
            cuentas = res.json().get("cuentas", [])
            for c in cuentas:
                if c.get("moneda") == "peso_Argentino":
                    return float(c.get("disponible", 0.0))
        return 0.0

    def comprar_cedear_ci(self, simbolo, cantidad, precio_limite):
        """Envía orden Límite en Contado Inmediato (CI) a BYMA."""
        url = f"{IOL_API_URL}/api/v2/operar/Comprar"
        payload = {
            "mercado": "bCBA",
            "simbolo": simbolo,
            "cantidad": int(cantidad),
            "precio": float(precio_limite),
            "plazo": "t0",  # t0 = Contado Inmediato (CI)
            "validez": "2026-12-31"
        }
        res = requests.post(url, json=payload, headers=self.get_headers())
        if res.status_code == 200:
            return True, res.json()
        return False, res.text
